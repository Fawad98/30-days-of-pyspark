// GroupBy & Aggregations in Action
day05_aggregations.py 
from pyspark.sql import SparkSession
from pyspark.sql import functions as F

spark = SparkSession.builder.appName("Day05").getOrCreate()

data = [
    (1, "Alice",   "Engineering", 95000.0),
    (2, "Bob",     "Marketing",   72000.0),
    (3, "Charlie", "Engineering", 88000.0),
    (4, "Diana",   "HR",          81000.0),
    (5, "Eve",     "Marketing",   67000.0),
    (6, "Frank",   "Engineering", 102000.0),
]
df = spark.createDataFrame(data, ["id", "name", "department", "salary"])

# ── Basic groupBy + agg ────────────────────────────────
df.groupBy("department").agg(
    F.count("id").alias("headcount"),
    F.avg("salary").alias("avg_salary"),
    F.max("salary").alias("max_salary"),
    F.min("salary").alias("min_salary"),
    F.sum("salary").alias("total_payroll")
).orderBy(F.col("avg_salary").desc()).show()

# ── Group by multiple columns ──────────────────────────
df.groupBy("department", "seniority").agg(
    F.count("*").alias("count")
).show()

# ── countDistinct: unique values in group ─────────────
df.groupBy("department").agg(
    F.countDistinct("name").alias("unique_employees")
).show()

# ── Global aggregation (no groupBy) ───────────────────
df.agg(
    F.avg("salary").alias("company_avg_salary"),
    F.sum("salary").alias("total_payroll")
).show()

# ── Pivot table ────────────────────────────────────────
# Rows = department, Columns = seniority level, Values = avg salary
df.withColumn("band",
    F.when(F.col("salary") >= 90000, "Senior").otherwise("Junior")
).groupBy("department") \
 .pivot("band", ["Junior", "Senior"]) \
 .agg(F.avg("salary")) \
 .show()

spark.stop()

// groupBy vs Window Functions
✦ Use groupBy when you want to collapse rows into one row per group — like SQL GROUP BY. The result has fewer rows than the input.
⚠️ Use Window Functions (coming Day 8) when you want to compute aggregates but keep all rows — like ranking employees within their department while still showing every employee. groupBy would lose the individual rows.
// Key Takeaways
  • Always use .agg() with .alias() — it keeps column names readable and avoids auto-generated names.
  • F.count("*") counts all rows including nulls; F.count("col") counts only non-null values.
  • You can groupBy multiple columns — just pass them as separate arguments.
  • Skip groupBy entirely for global aggregations — call df.agg() directly.
  • Pivot tables are powerful but expensive — avoid pivoting on high-cardinality columns.
